    function generatePentagonGraph() {
      const centerX = 400, centerY = 400;
      const radiusOuter = 340;
      const radiusMid = 220;
      const radiusInner = 110;

      let nodes = [];
      
      // Lớp 1: 5 Góc ngoài cùng (Spawn Points của 5 Team) -> ID 0..4
      for(let i=0; i<5; i++) {
        let angle = (i * 72 - 90) * Math.PI / 180;
        nodes.push({
          id: i,
          x: centerX + radiusOuter * Math.cos(angle),
          y: centerY + radiusOuter * Math.sin(angle),
          type: 'spawn',
          teamSpawn: i + 1,
          owner: i + 1, // Đã sở hữu sẵn bởi Đội xuất phát
          qId: null
        });
      }

      // Lớp 2: Ring giữa gồm 10 ô (5 đỉnh mid, 5 cạnh mid) -> ID 5..14
      for(let i=0; i<10; i++) {
        let angle = (i * 36 - 90) * Math.PI / 180;
        let isCam = (i % 2 === 0);
        nodes.push({
          id: 5 + i,
          x: centerX + radiusMid * Math.cos(angle),
          y: centerY + radiusMid * Math.sin(angle),
          type: isCam ? 'orange' : 'white',
          owner: null,
          qId: isCam ? getRandomQuestionId() : null
        });
      }

      // Lớp 3: Ring trong gồm 5 ô -> ID 15..19
      for(let i=0; i<5; i++) {
        let angle = (i * 72 - 90) * Math.PI / 180;
        let isCam = (i % 2 === 1);
        nodes.push({
          id: 15 + i,
          x: centerX + radiusInner * Math.cos(angle),
          y: centerY + radiusInner * Math.sin(angle),
          type: isCam ? 'orange' : 'white',
          owner: null,
          qId: isCam ? getRandomQuestionId() : null
        });
      }

      // Ô Trung Tâm -> ID 20
      nodes.push({
        id: 20,
        x: centerX,
        y: centerY,
        type: 'white',
        owner: null,
        qId: null
      });

      // Tạo ma trận liên kết giữa các ô liền kề
      let adj = {};
      for(let i=0; i<=20; i++) adj[i] = [];

      // Nối Lớp Outer (0..4) vào Lớp Mid
      adj[0].push(5, 6, 14); adj[5].push(0); adj[6].push(0); adj[14].push(0);
      adj[1].push(7, 6, 8);  adj[7].push(1); adj[6].push(1); adj[8].push(1);
      adj[2].push(9, 8, 10); adj[9].push(2); adj[8].push(2); adj[10].push(2);
      adj[3].push(11, 10, 12); adj[11].push(3); adj[10].push(3); adj[12].push(3);
      adj[4].push(13, 12, 14); adj[13].push(4); adj[12].push(4); adj[14].push(4);

      // Nối Vòng Mid (5..14)
      for(let i=5; i<=14; i++) {
        let next = i === 14 ? 5 : i + 1;
        adj[i].push(next);
        adj[next].push(i);
      }

      // Nối Mid vào Inner & Tâm
      adj[5].push(15); adj[15].push(5);
      adj[7].push(16); adj[16].push(7);
      adj[9].push(17); adj[17].push(9);
      adj[11].push(18); adj[18].push(11);
      adj[13].push(19); adj[19].push(13);

      // Nối Vòng Inner (15..19)
      for(let i=15; i<=19; i++) {
        let next = i === 19 ? 15 : i + 1;
        adj[i].push(next);
        adj[next].push(i);
        adj[i].push(20);
        adj[20].push(i);
      }

      // Loại bỏ trùng lặp trong adj
      for(let k in adj) {
        adj[k] = [...new Set(adj[k])];
      }

      boardNodes = nodes;
      boardConnections = adj;
    }

    let usedQuestionIds = [];
    function getRandomQuestionId() {
      let pool = QUESTIONS_DATA.map(q => q.id).filter(id => !usedQuestionIds.includes(id));
      if(pool.length === 0) pool = QUESTIONS_DATA.map(q => q.id);
      let picked = pool[Math.floor(Math.random() * pool.length)];
      usedQuestionIds.push(picked);
      return picked;
    }

    // --- RENDER BÀN CỜ SVG ---
    function renderBoard() {
      const svg = document.getElementById('game-board');
      svg.innerHTML = '';

      // 1. Vẽ các đường nối giữa ô liền kề
      let drawnLines = new Set();
      for(let u in boardConnections) {
        boardConnections[u].forEach(v => {
          let pairKey = [u, v].sort().join('-');
          if(!drawnLines.has(pairKey)) {
            drawnLines.add(pairKey);
            let n1 = boardNodes[u];
            let n2 = boardNodes[v];
            let line = document.createElementNS('http://www.w3.org/2000/svg', 'line');
            line.setAttribute('x1', n1.x);
            line.setAttribute('y1', n1.y);
            line.setAttribute('x2', n2.x);
            line.setAttribute('y2', n2.y);
            line.setAttribute('stroke', '#334155');
            line.setAttribute('stroke-width', '4');
            svg.appendChild(line);
          }
        });
      }

      // 2. Tìm danh sách các ô hợp lệ để di chuyển cho Đội hiện tại
      let activeTeam = TEAMS[currentTeamIdx];
      let validNeighbors = boardConnections[activeTeam.currentNode];

      // 3. Vẽ các Ô hình tròn
      boardNodes.forEach(node => {
        let group = document.createElementNS('http://www.w3.org/2000/svg', 'g');
        group.setAttribute('class', 'node');
        
        let circle = document.createElementNS('http://www.w3.org/2000/svg', 'circle');
        circle.setAttribute('cx', node.x);
        circle.setAttribute('cy', node.y);
        circle.setAttribute('r', '22');

        // Xác định màu tô cho Ô
        let fillColor = '#ffffff'; // Mặc định ô trắng
        if(node.owner) {
          fillColor = TEAMS.find(t => t.id === node.owner).color;
        } else if(node.type === 'orange') {
          fillColor = '#f59e0b'; // Ô cam câu hỏi
        }

        circle.setAttribute('fill', fillColor);
        circle.setAttribute('stroke', '#1e293b');
        circle.setAttribute('stroke-width', '3');

        // Hiệu ứng gợi ý ô có thể bước tới
        let isSelectable = validNeighbors.includes(node.id);
        if(isSelectable) {
          circle.setAttribute('stroke', '#38bdf8');
          circle.setAttribute('stroke-width', '5');
          circle.setAttribute('stroke-dasharray', '4');
          group.classList.add('selectable');
          group.onclick = () => handleMoveToNode(node.id);
        }

        group.appendChild(circle);

        // Icon hoặc nhãn ô
        if(node.type === 'orange' && !node.owner) {
          let text = document.createElementNS('http://www.w3.org/2000/svg', 'text');
          text.setAttribute('x', node.x);

          text.setAttribute('text-anchor', 'middle');
          text.setAttribute('fill', '#fff');
          text.setAttribute('font-weight', '900');
          text.setAttribute('font-size', '14');
          text.textContent = '?';
          group.appendChild(text);
        }

        svg.appendChild(group);
      });

      // 4. Vẽ Quân cờ đại diện cho 5 Đội
      TEAMS.forEach(team => {
        let posNode = boardNodes[team.currentNode];

        token.setAttribute('cx', posNode.x);
        token.setAttribute('cy', posNode.y);
        token.setAttribute('r', '12');
        token.setAttribute('fill', team.color);
        token.setAttribute('stroke', '#ffffff');
        token.setAttribute('stroke-width', '3');

        text.setAttribute('font-weight', '900');
        text.setAttribute('font-size', '11');
        text.textContent = team.id;

        svg.appendChild(token);
        svg.appendChild(text);
      });

    // --- XỬ LÝ DI CHUYỂN TỚI Ô MỚI ---
    function handleMoveToNode(targetNodeId) {
      if(isGameOver) return;
      clearInterval(turnTimerInterval);

      let team = TEAMS[currentTeamIdx];
      team.currentNode = targetNodeId;
      let targetNode = boardNodes[targetNodeId];

      addLog(`<b>${team.name}</b> đã di chuyển tới ô <b>#${targetNodeId}</b>.`);

      // 1. Kiểm tra dẫm vào ô màu của Đội khác -> Nộp phí -3 điểm
      if(targetNode.owner && targetNode.owner !== team.id) {
        let ownerTeam = TEAMS.find(t => t.id === targetNode.owner);
        team.score -= 3;
        ownerTeam.score += 3;
        addLog(`<span style="color:#ef4444;"><i class="fa-solid fa-coins"></i> ${team.name} dẫm vào ô của ${ownerTeam.name}! Nộp phí -3 điểm.</span>`);
      }

      // Tô màu ô cho đội đi vào (nếu chưa có chủ)
      if(!targetNode.owner) {
        targetNode.owner = team.id;
        addLog(`<span style="color:${team.color};"><i class="fa-solid fa-flag"></i> ${team.name} đã chiếm đóng ô #${targetNodeId}!</span>`);
      }

      updateTeamsUI();

      // 2. Kích hoạt sự kiện Ô Cam (Câu Hỏi) hoặc Ô Trắng (Sự Kiện Ngẫu Nhiên)
      if(targetNode.type === 'orange') {
        triggerQuestionModal(targetNode);
      } else {
        triggerRandomEventModal(targetNode);
      }
    }

    window.handleChoiceOption = function(type, targetTeamId) {

      } else {
        let prevTeam = TEAMS.find(t => t.id === targetTeamId);
        let deductAmount = Math.floor(prevTeam.score / 2);
        prevTeam.score -= deductAmount;
        addLog(`[Lựa chọn] ${activeTeam.name} trừ ${deductAmount} điểm (nửa số điểm) của ${prevTeam.name}!`);
      }
      updateTeamsUI();
      closeModalNextTurn();
    };

    window.submitAnswer = function(type, qId, userAns) {
      let q = QUESTIONS_DATA.find(item => item.id === qId);
      let activeTeam = TEAMS[currentTeamIdx];
      let isCorrect = false;

      if(type === 'fill') {
        let inputVal = document.getElementById('fill-answer').value.trim().toLowerCase();
        let expected = q.answer.toLowerCase();
        isCorrect = inputVal.includes(expected) || expected.includes(inputVal) && inputVal.length >= 3;
      } else if(type === 'tf') {
        isCorrect = (userAns === q.answer);
